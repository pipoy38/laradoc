# Laradoc Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Wiki.js-style documentation site: public reader front end, password-protected admin with articles, nested categories, revisions, media, menu, users and settings, deployable to any PHP host.

**Architecture:** Plain Laravel 12 MVC with Blade views. A handful of small `App\Support` service classes (settings cache, admin prefix, HTML sanitiser, path resolver, search, TOC) hold the logic that isn't CRUD. Front routes resolve `/{category/path}/{article}` through one catch-all controller; admin routes live under a prefix read from the settings table at boot.

**Tech Stack:** PHP ≥8.2, Laravel 12, MySQL 8 (SQLite in tests), Blade, Tailwind CSS v4, Alpine.js 3, TinyMCE 8 (self-hosted, GPL), highlight.js 11, Vite, Pest.

**Spec:** `docs/superpowers/specs/2026-09-03-laradoc-design.md`

## Global Constraints

- Composer `config.platform.php` = `8.2.0`; `require.php` = `^8.2`. Laravel pinned `^12.0`.
- Composer deps: `laravel/framework`, `laravel/tinker`; dev: `pestphp/pest`, `pestphp/pest-plugin-laravel`, `laravel/pint`, `fakerphp/faker`, `mockery/mockery`, `nunomaduro/collision`. Nothing else.
- npm deps: `tailwindcss`, `@tailwindcss/vite`, `vite`, `laravel-vite-plugin`, `alpinejs`, `tinymce`, `highlight.js`. Nothing else.
- `public/build` is committed. No CDN, no web fonts, no API keys.
- Roles: `admin`, `editor`. No public login.
- Admin prefix default `admin`, validated `[a-z0-9-]{2,32}`.
- Reserved top-level paths: `search`, `tags`, `api`, `sitemap.xml`, `robots.txt`, `storage`, `build`, `vendor`, plus the admin prefix.
- Theme keys in localStorage: `laradoc.theme.front`, `laradoc.theme.admin`. Theme applied via `data-theme="dark"` on `<html>`.
- Palette (from ERP): brand `#4fb9d4`, brand-deep `#1c7f9b`, ink `#0b1228`, paper `#eef2f8`, sheet `#ffffff`, rule `#dfe5ee`; dark: paper `#0a0f1c`, sheet `#111828`, ink `#eef1f7`, rule `#2c3852`, brand `#62c8e0`, brand-deep `#8fd9ea`.

```js
import Alpine from 'alpinejs';
import hljs from 'highlight.js/lib/common';
import 'highlight.js/styles/github-dark.css';

window.Alpine = Alpine;

// Live search dropdown
Alpine.data('siteSearch', () => ({
    q: '', results: [], open: false, timer: null, active: -1,
    input() {
        clearTimeout(this.timer);
        if (this.q.trim().length < 2) { this.results = []; this.open = false; return; }
        this.timer = setTimeout(() => this.fetch(), 250);
    },
    async fetch() {
        const r = await fetch(`/api/search?q=${encodeURIComponent(this.q)}`, { headers: { Accept: 'application/json' } });
        this.results = r.ok ? await r.json() : [];
        this.open = this.results.length > 0; this.active = -1;
    },
    move(d) { if (!this.open) return; this.active = Math.max(-1, Math.min(this.results.length - 1, this.active + d)); },
    go() { if (this.active >= 0) { window.location = this.results[this.active].url; return false; } return true; },
}));

// Scroll-spy for "On this page"
Alpine.data('toc', () => ({
    current: '',
    init() {
        const heads = [...document.querySelectorAll('.prose-doc h2[id], .prose-doc h3[id]')];
        if (!heads.length) return;
        const io = new IntersectionObserver((entries) => {
            entries.forEach(e => { if (e.isIntersecting) this.current = e.target.id; });
        }, { rootMargin: '-80px 0px -70% 0px' });
        heads.forEach(h => io.observe(h));
    },
}));

Alpine.start();

// Code blocks: highlight + copy
document.querySelectorAll('.prose-doc pre code').forEach((code) => {
    hljs.highlightElement(code);
    const btn = document.createElement('button');
    btn.type = 'button'; btn.className = 'copy-btn'; btn.textContent = 'Copy';
    btn.addEventListener('click', async () => {
        await navigator.clipboard.writeText(code.innerText);
        btn.textContent = 'Copied'; setTimeout(() => (btn.textContent = 'Copy'), 1500);
    });
    code.parentElement.appendChild(btn);
});
// Wrap tables for horizontal scroll
document.querySelectorAll('.prose-doc table').forEach((t) => {
    const w = document.createElement('div'); w.className = 'table-wrap';
    t.parentNode.insertBefore(w, t); w.appendChild(t);
```

```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Collection;

class Category extends Model
{
    use HasFactory;
    protected $fillable = ['parent_id', 'name', 'slug', 'description', 'icon', 'sort_order'];

    public function parent() { return $this->belongsTo(self::class, 'parent_id'); }
    public function children() { return $this->hasMany(self::class, 'parent_id')->ordered(); }
    public function articles() { return $this->hasMany(Article::class)->orderBy('sort_order')->orderBy('title'); }
    public function scopeOrdered(Builder $q): Builder { return $q->orderBy('sort_order')->orderBy('name'); }

    /** Root-first chain of ancestors, excluding self. */
    public function ancestors(): Collection
    {
        $chain = collect(); $node = $this->parent;
        while ($node) { $chain->prepend($node); $node = $node->parent; }
        return $chain;
    }

    public function path(): string
    {
        return $this->ancestors()->push($this)->pluck('slug')->implode('/');
    }

    public function url(): string { return url('/'.$this->path()); }

    /** Self plus every descendant id, depth-first. */
    public function descendantIds(): array
    {
        $ids = [$this->id];
        foreach ($this->children as $child) { $ids = array_merge($ids, $child->descendantIds()); }
        return $ids;
    }

    /** True when $candidate is this category or one of its descendants (used to forbid cyclic parents). */
    public function isOrContains(?int $candidateId): bool
    {
        return $candidateId !== null && in_array($candidateId, $this->descendantIds(), true);
    }
}
```

```php
        if ($value === null || $value === '') { return $default ?? (self::DEFAULTS[$key] ?? null); }
        if (in_array($key, self::ENCRYPTED, true)) { try { return Crypt::decryptString($value); } catch (\Throwable) { return null; } }
        return $value;
    }

    public function set(string $key, ?string $value): void { $this->setMany([$key => $value]); }

    public function setMany(array $pairs): void
    {
        foreach ($pairs as $key => $value) {
            if ($value !== null && in_array($key, self::ENCRYPTED, true)) { $value = Crypt::encryptString($value); }
            Setting::updateOrCreate(['key' => $key], ['value' => $value]);
        }
        $this->flush();
    }

    public function flush(): void { $this->loaded = null; Cache::forget(self::CACHE_KEY); }
}
```

`app/Support/PathResolver.php`:
```php
<?php
namespace App\Support;

use App\Models\Article;
use App\Models\Category;
use App\Models\PathRedirect;
use Illuminate\Support\Collection;

class PathResolver
{
    public function resolve(string $path, bool $publishedOnly = true): ?array
    {
        $segments = array_values(array_filter(explode('/', trim($path, '/')), fn ($s) => $s !== ''));
        if ($segments === []) { return null; }

        // Walk categories as far as the path allows.
        $parentId = null; $category = null; $consumed = 0;
        foreach ($segments as $slug) {
            $next = Category::where('parent_id', $parentId)->where('slug', $slug)->first();
            if (! $next) { break; }
            $category = $next; $parentId = $next->id; $consumed++;
        }
        $remaining = array_slice($segments, $consumed);

        if ($remaining === [] && $category) { return ['type' => 'category', 'category' => $category]; }
        if (count($remaining) === 1 && $category) {
            $q = Article::where('category_id', $category->id)->where('slug', $remaining[0]);
            if ($publishedOnly) { $q->published(); }
            if ($article = $q->first()) { return ['type' => 'article', 'article' => $article]; }
        }
        return null;
    }

    /** Root categories with nested children and published articles, ordered. */
    public function tree(): Collection
    {
        $categories = Category::ordered()->get();
        $articles = Article::published()->orderBy('sort_order')->orderBy('title')->get(['id', 'category_id', 'title', 'slug'])->groupBy('category_id');
        $byParent = $categories->groupBy(fn ($c) => $c->parent_id ?? 0);
        $build = function ($parentId) use (&$build, $byParent, $articles) {
            return ($byParent[$parentId] ?? collect())->each(function ($c) use (&$build, $articles) {
                $c->setRelation('children', $build($c->id));
                $c->setRelation('articles', $articles[$c->id] ?? collect());
            })->values();
        };
        return $build(0);
    }

    /** Record 301s for every public path under $category (or the article itself) when its path changes. */
    public function recordRedirects(string $oldPath, string $newPath, ?Category $category = null): void
    {
        if ($oldPath === $newPath) { return; }
        $this->redirect($oldPath, $newPath);
        if ($category) {
            foreach (Category::whereIn('id', $category->descendantIds())->get() as $cat) {
                $catNew = $cat->path(); $catOld = $oldPath.substr($catNew, strlen($newPath));
                if ($cat->id !== $category->id) { $this->redirect($catOld, $catNew); }
                foreach ($cat->articles as $article) { $this->redirect($catOld.'/'.$article->slug, $catNew.'/'.$article->slug); }
            }
        }
    }

    private function redirect(string $from, string $to): void
    {
        if ($from === $to) { return; }
        PathRedirect::where('from_path', $to)->delete(); // a path that is live again must not redirect away
        PathRedirect::updateOrCreate(['from_path' => $from], ['to_path' => $to]);
        PathRedirect::where('to_path', $from)->update(['to_path' => $to]); // chain-collapse older redirects
    }
}
```

```php
use App\Http\Requests\Admin\CategoryRequest;
use App\Models\Category;
use App\Support\PathResolver;
use Illuminate\Http\Request;

class CategoryController extends Controller
{
    public function index()
    {
        return view('admin.categories.index', ['tree' => self::adminTree()]);
    }

    public function create() { return view('admin.categories.form', ['category' => new Category(), 'parentId' => request()->integer('parent') ?: null]); }

    public function store(CategoryRequest $request)
    {
        $data = $request->validated();
        $data['sort_order'] = (Category::where('parent_id', $data['parent_id'])->max('sort_order') ?? -1) + 1;
        Category::create($data);
        return redirect()->route('admin.categories.index')->with('ok', 'Category created.');
    }

    public function edit(Category $category) { return view('admin.categories.form', ['category' => $category, 'parentId' => $category->parent_id]); }

    public function update(CategoryRequest $request, Category $category, PathResolver $resolver)
    {
        $oldPath = $category->path();
        $category->fill($request->validated());
        if ($category->isDirty('parent_id')) { $category->sort_order = (Category::where('parent_id', $category->parent_id)->max('sort_order') ?? -1) + 1; }
        $category->save();
        $category->refresh()->load('parent');
        $resolver->recordRedirects($oldPath, $category->path(), $category);
        return redirect()->route('admin.categories.index')->with('ok', 'Category updated.');
    }

    public function destroy(Category $category)
    {
        if ($category->children()->exists() || $category->articles()->exists()) {
            return back()->with('err', 'Move or delete its sub-categories and articles first.');
        }
        $category->delete();
        return redirect()->route('admin.categories.index')->with('ok', 'Category deleted.');
    }

    public function move(Request $request, Category $category)
    {
        $dir = $request->input('dir') === 'up' ? -1 : 1;
        $siblings = Category::where('parent_id', $category->parent_id)->ordered()->get()->values();
        $i = $siblings->search(fn ($c) => $c->id === $category->id);
        $j = $i + $dir;
        if (isset($siblings[$j])) {
            foreach ($siblings as $k => $s) { $s->sort_order = $k; }
            [$siblings[$i]->sort_order, $siblings[$j]->sort_order] = [$j, $i];
            $siblings->each->save();
        }
        return back();
    }

    /** All categories nested (drafts included) with article counts, for admin screens. */
    public static function adminTree(): \Illuminate\Support\Collection
    {
        $all = Category::ordered()->withCount('articles')->get();
        $byParent = $all->groupBy(fn ($c) => $c->parent_id ?? 0);
        $build = function ($parentId, $depth = 0) use (&$build, $byParent) {
            return ($byParent[$parentId] ?? collect())->each(function ($c) use (&$build, $depth) { $c->depth = $depth; $c->setRelation('children', $build($c->id, $depth + 1)); })->values();
        };
        return $build(0);
    }

    /** Flatten adminTree() into [id => ['name', 'depth', 'path']] for selects. */
    public static function flatOptions(): array
    {
        $out = [];
        $walk = function ($nodes) use (&$walk, &$out) { foreach ($nodes as $n) { $out[$n->id] = ['name' => $n->name, 'depth' => $n->depth, 'path' => $n->path()]; $walk($n->children); } };
        $walk(self::adminTree());
        return $out;
    }
}
```

```php
<?php
namespace App\Support;

use Illuminate\Support\Str;

class HtmlSanitizer
{
    private const IFRAME_HOSTS = ['www.youtube.com', 'youtube.com', 'www.youtube-nocookie.com', 'player.vimeo.com', 'vimeo.com'];

    public static function clean(string $html): string
    {
        if (trim($html) === '') { return ''; }
        $doc = self::load($html);
        $xpath = new \DOMXPath($doc);
        foreach (['script', 'object', 'embed', 'style', 'form', 'input', 'button', 'meta', 'link', 'base'] as $tag) {
            foreach (iterator_to_array($doc->getElementsByTagName($tag)) as $el) { $el->parentNode?->removeChild($el); }
        }
        foreach (iterator_to_array($doc->getElementsByTagName('iframe')) as $el) {
            $host = strtolower((string) parse_url($el->getAttribute('src'), PHP_URL_HOST));
            if (! in_array($host, self::IFRAME_HOSTS, true)) { $el->parentNode?->removeChild($el); continue; }
            foreach (['srcdoc', 'onload'] as $a) { $el->removeAttribute($a); }
        }
        foreach ($xpath->query('//@*') as $attr) {
            $name = strtolower($attr->nodeName); $value = trim($attr->nodeValue ?? '');
            $isUrlAttr = in_array($name, ['href', 'src', 'xlink:href', 'action', 'formaction', 'poster', 'data'], true);
            if (str_starts_with($name, 'on') || ($isUrlAttr && preg_match('/^\s*(javascript|vbscript|data:text\/html)/i', $value))) {
                $attr->ownerElement?->removeAttribute($attr->nodeName);
            }
        }
        return self::innerHtml($doc);
    }

    public static function toText(string $html): string
    {
        $text = html_entity_decode(strip_tags(preg_replace('/<(\/?)(p|div|br|h[1-6]|li|tr|blockquote|pre)[^>]*>/i', ' ', $html) ?? ''), ENT_QUOTES | ENT_HTML5, 'UTF-8');
        return trim(preg_replace('/\s+/u', ' ', $text) ?? '');
    }

    public static function excerpt(string $html, int $len = 200): string { return Str::limit(self::toText($html), $len, '…'); }

    private static function load(string $html): \DOMDocument
    {
        $doc = new \DOMDocument('1.0', 'UTF-8');
        libxml_use_internal_errors(true);
        $doc->loadHTML('<?xml encoding="UTF-8"><div id="__root">'.$html.'</div>', LIBXML_HTML_NOIMPLIED | LIBXML_HTML_NODEFDTD | LIBXML_NONET);
        libxml_clear_errors();
        return $doc;
    }

    private static function innerHtml(\DOMDocument $doc): string
    {
        $root = $doc->getElementById('__root'); $out = '';
        foreach ($root->childNodes as $child) { $out .= $doc->saveHTML($child); }
        return trim($out);
    }
}
```

```php
<?php
namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\ArticleRequest;
use App\Models\Article;
use App\Models\Tag;
use App\Support\HtmlSanitizer;
use App\Support\PathResolver;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class ArticleController extends Controller
{
    public function index(Request $request)
    {
        $q = Article::with(['category', 'editor', 'creator'])->latest('updated_at');
        if ($s = $request->input('status')) { $q->where('status', $s); }
        if ($c = $request->integer('category')) { $q->where('category_id', $c); }
        if ($t = $request->input('tag')) { $q->whereHas('tags', fn ($x) => $x->where('slug', $t)); }
        if ($text = trim((string) $request->input('q'))) { $q->where(fn ($x) => $x->where('title', 'like', "%{$text}%")->orWhere('slug', 'like', "%{$text}%")); }
        return view('admin.articles.index', ['articles' => $q->paginate(25)->withQueryString(), 'options' => CategoryController::flatOptions(), 'tags' => Tag::orderBy('name')->get()]);
    }

    public function create() { return view('admin.articles.form', ['article' => new Article(['status' => 'draft']), 'tagString' => '']); }

    public function store(ArticleRequest $request)
    {
        $article = DB::transaction(function () use ($request) {
            $article = new Article($this->attributes($request));
            $article->created_by = $request->user()->id;
            $article->sort_order = (Article::where('category_id', $article->category_id)->max('sort_order') ?? -1) + 1;
            $article->save();
            $this->syncTags($article, $request->input('tags'));
            $this->snapshot($article, $request);
            return $article;
        });
        return redirect()->route('admin.articles.edit', $article)->with('ok', 'Article created.');
    }

    public function edit(Article $article)
    {
        $article->load(['tags', 'revisions.user']);
        return view('admin.articles.form', ['article' => $article, 'tagString' => $article->tags->pluck('name')->implode(', ')]);
    }

    public function update(ArticleRequest $request, Article $article, PathResolver $resolver)
    {
        DB::transaction(function () use ($request, $article, $resolver) {
            $oldPath = $article->path();
            $article->fill($this->attributes($request, $article));
            $article->updated_by = $request->user()->id;
            if ($article->isDirty('category_id')) { $article->sort_order = (Article::where('category_id', $article->category_id)->max('sort_order') ?? -1) + 1; }
            $article->save();
            $article->unsetRelation('category');
            $resolver->recordRedirects($oldPath, $article->path());
            $this->syncTags($article, $request->input('tags'));
            $this->snapshot($article, $request);
        });
        return redirect()->route('admin.articles.edit', $article)->with('ok', 'Saved.');
    }

    public function destroy(Article $article)
    {
        $article->delete();
        return redirect()->route('admin.articles.index')->with('ok', 'Article deleted.');
    }

    private function attributes(ArticleRequest $request, ?Article $existing = null): array
    {
        $data = $request->safe()->except(['tags', 'change_note']);
        $data['body_html'] = HtmlSanitizer::clean($data['body_html']);
        $data['body_text'] = HtmlSanitizer::toText($data['body_html']);
        $data['excerpt'] = trim((string) ($data['excerpt'] ?? '')) ?: HtmlSanitizer::excerpt($data['body_html']);
        if ($data['status'] === 'published' && ! $existing?->published_at) { $data['published_at'] = now(); }
        return $data;
    }

    private function syncTags(Article $article, ?string $csv): void
    {
        $ids = collect(explode(',', (string) $csv))->map(fn ($t) => trim($t))->filter()->unique(fn ($t) => Str::slug($t))
            ->map(fn ($name) => Tag::firstOrCreate(['slug' => Str::slug($name)], ['name' => $name])->id)->all();
        $article->tags()->sync($ids);
    }

    private function snapshot(Article $article, Request $request, ?string $note = null): void
    {
        $article->revisions()->create(['user_id' => $request->user()->id, 'title' => $article->title, 'body_html' => $article->body_html, 'excerpt' => $article->excerpt,
            'seo_title' => $article->seo_title, 'meta_description' => $article->meta_description, 'change_note' => $note ?? $request->input('change_note'), 'created_at' => now()]);
    }
}
```

```php
namespace App\Http\Controllers\Admin;
```

```php
        return view('admin.media.index', ['media' => Media::with('uploader')->latest()->paginate(40)]);

    public function upload(Request $request)
    {
        $request->validate(['file' => 'required|file|max:5120|mimes:jpg,jpeg,png,gif,webp,svg|mimetypes:image/jpeg,image/png,image/gif,image/webp,image/svg+xml']);
        $file = $request->file('file');
        $ext = strtolower($file->getClientOriginalExtension() ?: $file->extension());
        $dir = 'media/'.now()->format('Y/m');
        $name = Str::random(16).'.'.$ext;
        if ($ext === 'svg') {
            $svg = preg_replace(['/<script\b[^>]*>.*?<\/script>/is', '/<foreignObject\b[^>]*>.*?<\/foreignObject>/is', '/\son\w+="[^"]*"/i'], '', file_get_contents($file->getRealPath()));
            Storage::disk('public')->put("$dir/$name", $svg);
            [$w, $h] = [null, null];
        } else {
            $path = $file->storeAs($dir, $name, 'public');
            $size = @getimagesize(Storage::disk('public')->path($path));
            [$w, $h] = $size ? [$size[0], $size[1]] : [null, null];
        }
        $media = Media::create(['disk_path' => "$dir/$name", 'original_name' => $file->getClientOriginalName(), 'mime' => $file->getMimeType(), 'size' => $file->getSize(),
            'width' => $w, 'height' => $h, 'uploaded_by' => $request->user()->id]);
        if ($request->expectsJson()) { return response()->json(['location' => $media->url(), 'id' => $media->id], 201); }
        return redirect()->route('admin.media.index')->with('ok', 'Uploaded.');
    }

    public function update(Request $request, Media $media)
    {
        $media->update($request->validate(['alt' => 'nullable|string|max:255']));
        return back()->with('ok', 'Alt text saved.');
    }

    public function destroy(Media $media)
    {
        $inUse = in_array((string) $media->id, [setting('logo_media_id'), setting('favicon_media_id')], true) || Article::where('og_image_id', $media->id)->exists();
        if ($inUse) { return back()->with('err', 'This file is used as the logo, favicon or a share image. Change that first.'); }
        Storage::disk('public')->delete($media->disk_path);
        $media->delete();
        return back()->with('ok', 'File deleted.');
    }
}
```

```php
<?php
namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\MenuItem;
use Illuminate\Http\Request;

class MenuItemController extends Controller
{
    private function rules(): array { return ['label' => 'required|string|max:60', 'url' => 'required|string|max:500', 'new_tab' => 'nullable|boolean']; }

    public function index() { return view('admin.menu.index', ['items' => MenuItem::orderBy('sort_order')->get()]); }

    public function store(Request $request)
    {
        $data = $request->validate($this->rules());
        MenuItem::create($data + ['new_tab' => $request->boolean('new_tab'), 'sort_order' => (MenuItem::max('sort_order') ?? -1) + 1]);
        return redirect()->route('admin.menu.index')->with('ok', 'Menu item added.');
    }

    public function update(Request $request, MenuItem $menu)
    {
        $menu->update($request->validate($this->rules()) + ['new_tab' => $request->boolean('new_tab')]);
        return redirect()->route('admin.menu.index')->with('ok', 'Menu item saved.');
    }

    public function destroy(MenuItem $menu) { $menu->delete(); return back()->with('ok', 'Menu item removed.'); }

    public function move(Request $request, MenuItem $menu)
    {
        $items = MenuItem::orderBy('sort_order')->get()->values();
        $i = $items->search(fn ($m) => $m->id === $menu->id); $j = $i + ($request->input('dir') === 'up' ? -1 : 1);
        if (isset($items[$j])) { foreach ($items as $k => $m) { $m->sort_order = $k; } [$items[$i]->sort_order, $items[$j]->sort_order] = [$j, $i]; $items->each->save(); }
        return back();
    }
}
```

```php
<?php
namespace App\Support;

use App\Models\Article;
use Illuminate\Support\Collection;
use Illuminate\Support\Facades\DB;

class SearchService
{
    public function search(string $q, int $limit = 20): Collection
    {
        $q = trim(preg_replace('/\s+/', ' ', $q) ?? '');
        if (mb_strlen($q) < 2) { return collect(); }
        $query = Article::published()->with('category');
        if (DB::connection()->getDriverName() === 'mysql') {
            $boolean = collect(explode(' ', $q))->map(fn ($w) => '+'.preg_replace('/[+\-<>()~*"@]/', '', $w).'*')->implode(' ');
            $query->whereRaw('MATCH(title, excerpt, body_text) AGAINST(? IN BOOLEAN MODE)', [$boolean])
                  ->orderByRaw('MATCH(title, excerpt, body_text) AGAINST(? IN BOOLEAN MODE) DESC', [$boolean]);
        } else {
            foreach (explode(' ', $q) as $word) {
                $query->where(fn ($x) => $x->where('title', 'like', "%$word%")->orWhere('excerpt', 'like', "%$word%")->orWhere('body_text', 'like', "%$word%"));
            }
            $query->orderByRaw("CASE WHEN title LIKE ? THEN 0 ELSE 1 END, title", ["%$q%"]);
        }
        return $query->limit($limit)->get();
    }

    public function snippet(Article $a, string $q, int $len = 160): string
    {
        $text = $a->body_text; $pos = mb_stripos($text, explode(' ', trim($q))[0] ?? '');
        $start = $pos === false ? 0 : max(0, $pos - 60);
        return ($start > 0 ? '…' : '').mb_substr($text, $start, $len).(mb_strlen($text) > $start + $len ? '…' : '');
    }
}
```
